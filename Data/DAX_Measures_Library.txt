// ===============================================================================
// 1. WORKFORCE HEADCOUNT & ACTIVE CAPACITY
// ===============================================================================

Total Headcount = 
CALCULATE(
    COUNT(Employee_Master[Employee_ID]),
    FILTER(
        ALL(Employee_Master),
        Employee_Master[Hire_Date] <= MAX(Calendar[Date])
        &&
        (
            ISBLANK(Employee_Master[Termination_Date])
            ||
            Employee_Master[Termination_Date] > MAX(Calendar[Date])
        )
    )
)

Calibrated_Headcount = 
CALCULATE(
    COUNTROWS('Strategic_Workforce_Dataset'),
    'Strategic_Workforce_Dataset'[Employment_Status] = "Active"
)

// ===============================================================================
// 2. ATTRITION, TENURE & PERFORMANCE BIAS ANALYTICS
// ===============================================================================

Voluntary Attrition Rate % = 
VAR TotalVoluntaryLeavers = 
    CALCULATE(
        COUNTROWS(Employee_Master),
        Employee_Master[Termination_Type] = "Voluntary",
        USERELATIONSHIP(Calendar[Date], Employee_Master[Termination_Date])
    )
RETURN
    DIVIDE(TotalVoluntaryLeavers, [Total Headcount], 0)

High_Performers_Pct = 
VAR HighPerformers = 
    CALCULATE(
        [Calibrated_Headcount],
        'Strategic_Workforce_Dataset'[Performance_Rating] IN {"Exceeds Expectations", "Top Performer"}
    )
RETURN
DIVIDE(HighPerformers, [Calibrated_Headcount], 0)

Calibration_Bias_Risk_Flag = 
IF(
    [High_Performers_Pct] > 0.20,
    "HIGH RISK: Skewed Rating Distribution (>20%)",
    "GOVERNED: Normal Distribution Curve"
)

// ===============================================================================
// 3. HRIS DATA GOVERNANCE & COST BASELINE
// ===============================================================================

HRIS Data Integrity SLA % = 
VAR CleanRecords = 
    CALCULATE(
        COUNTROWS(Employee_Master),
        Employee_Master[Data_Quality_Flag] = "Clean"
    )
VAR TotalRecords = COUNTROWS(Employee_Master)
RETURN
    DIVIDE(CleanRecords, TotalRecords, 0)

Total Salary Cost Baseline = 
CALCULATE(
    SUM(Employee_Master[Base_Salary]),
    Employee_Master[Status] = "Active"
)
